"""告警评估引擎。

**两种评估方式缺一不可**：

  · **消息驱动** —— 字段越界类规则（电量低、温度高）。来了消息就评估一次，
    并按规则里的 `for` 做防抖（条件需持续满足 N 秒才触发，避免瞬时抖动误报）。
  · **定时扫描** —— 离线类规则。「一段时间没有消息」本质上是**没有事件发生**，
    没有消息就没有东西可驱动评估，只能靠定时扫描。

只做前者会漏掉离线；只做后者会漏掉字段越界。
"""

import json
import logging
import os
import threading
import time

log = logging.getLogger(__name__)

# 比较符
_OPS = {
    "lt": lambda v, t: v < t,
    "lte": lambda v, t: v <= t,
    "gt": lambda v, t: v > t,
    "gte": lambda v, t: v >= t,
    "eq": lambda v, t: v == t,
    "ne": lambda v, t: v != t,
}

OFFLINE_FIELD = "@offline"


class AlertEngine:
    def __init__(self, hub, store, rules, on_event=None, offline_scan_interval=3.0):
        self.hub = hub
        self.store = store
        self.rules = [r for r in (rules or []) if r.get("id")]
        self.on_event = on_event
        self.offline_scan_interval = offline_scan_interval

        self._pending = {}      # (ruleId, twinId) -> 首次满足条件的时间戳
        self._active = {}       # (ruleId, twinId) -> alertId
        self._last_seen = {}    # twinId -> 最后一条消息的时间戳
        self._stop = threading.Event()
        self._thread = None

    # ---- 生命周期 ----
    def start(self):
        # 用当前已知设备做初值，避免刚启动就把所有设备判为离线
        now = int(time.time() * 1000)
        for twin in self.hub.snapshot():
            self._last_seen[twin["twinId"]] = now

        self._thread = threading.Thread(target=self._run, daemon=True)
        self._thread.start()
        log.info("告警引擎已启动：%d 条规则（%d 条字段规则 + %d 条离线规则）",
                 len(self.rules),
                 sum(1 for r in self.rules if r.get("field") != OFFLINE_FIELD),
                 sum(1 for r in self.rules if r.get("field") == OFFLINE_FIELD))

    def stop(self):
        self._stop.set()

    def _run(self):
        while not self._stop.wait(self.offline_scan_interval):
            try:
                self.evaluate_offline()
            except Exception as exc:  # noqa: BLE001
                log.warning("离线扫描出错：%s", exc)

    # ---- 消息驱动 ----
    def evaluate_message(self, msg):
        """由 Hub 在每条入站消息后调用。"""
        twin_id = msg.get("twinId")
        if not twin_id:
            return
        now = int(time.time() * 1000)
        self._last_seen[twin_id] = now

        data = msg.get("data") or {}
        for rule in self.rules:
            if rule.get("field") == OFFLINE_FIELD:
                continue  # 离线规则交给定时扫描
            if not self._matches_twin(rule, twin_id):
                continue
            value = data.get(rule.get("field"))
            if not isinstance(value, (int, float)):
                continue

            key = (rule["id"], twin_id)
            if self._satisfied(value, rule):
                first = self._pending.setdefault(key, now)
                if now - first >= rule.get("for", 0) * 1000:
                    self._fire(rule, twin_id, value)
            else:
                # 条件不再满足：清掉防抖计时，并自动恢复已触发的告警
                self._pending.pop(key, None)
                self._resolve(rule, twin_id)

    def evaluate_offline(self):
        """定时扫描：检查是否有设备停止上报。"""
        now = int(time.time() * 1000)
        for rule in self.rules:
            if rule.get("field") != OFFLINE_FIELD:
                continue
            for twin_id, last in list(self._last_seen.items()):
                if not self._matches_twin(rule, twin_id):
                    continue
                gap = (now - last) / 1000
                if gap > rule["threshold"]:
                    self._fire(rule, twin_id, round(gap, 1))

    # ---- 触发 / 恢复 ----
    def _fire(self, rule, twin_id, value):
        key = (rule["id"], twin_id)
        if key in self._active:
            return  # 已在告警中，不重复产生
        message = format_message(rule, twin_id, value)
        alert = self.store.raise_alert(
            rule["id"], rule.get("name", rule["id"]), twin_id,
            rule.get("level", "warning"), message, value)
        self._active[key] = alert["id"]
        log.info("告警触发 [%s] %s：%s", alert["level"], twin_id, message)
        self._emit("fired", alert)

    def _resolve(self, rule, twin_id):
        key = (rule["id"], twin_id)
        alert_id = self._active.pop(key, None)
        if alert_id is None:
            return
        self.store.resolve(alert_id)
        log.info("告警恢复 [%s] %s", rule["id"], twin_id)
        self._emit("resolved", {"id": alert_id, "ruleId": rule["id"], "twinId": twin_id})

    def _emit(self, action, alert):
        if self.on_event:
            try:
                self.on_event(action, alert)
            except Exception as exc:  # noqa: BLE001
                log.warning("告警通知回调出错：%s", exc)

    # ---- 工具 ----
    @staticmethod
    def _matches_twin(rule, twin_id):
        target = rule.get("twinId", "*")
        return target == "*" or target == twin_id

    @staticmethod
    def _satisfied(value, rule):
        op = _OPS.get(rule.get("op", "lt"))
        return bool(op and op(value, rule["threshold"]))

    # ---- 测试/调试用 ----
    def state(self):
        return {
            "rules": len(self.rules),
            "active": len(self._active),
            "pending": len(self._pending),
            "trackedTwins": len(self._last_seen),
        }


def format_message(rule, twin_id, value):
    """把规则里的占位符填上。缺字段时保留模板，不至于报错。"""
    template = rule.get("message") or rule.get("name") or rule["id"]
    try:
        return template.format(value=value, threshold=rule.get("threshold"),
                               twinId=twin_id, name=rule.get("name", ""))
    except (KeyError, IndexError, ValueError):
        return template


def load_rules(path):
    """从 JSON 载入规则表；文件缺失或损坏时返回空表而不是让服务起不来。"""
    if not os.path.exists(path):
        log.warning("告警规则文件不存在：%s", path)
        return []
    try:
        with open(path, encoding="utf-8") as fh:
            data = json.load(fh)
        rules = data.get("rules") if isinstance(data, dict) else data
        return rules or []
    except (OSError, ValueError) as exc:
        log.warning("告警规则载入失败（%s）：%s", path, exc)
        return []

"""告警的持久化与查询。

单独一个库文件：告警是**事件流**，访问模式（按时间倒序翻、按级别筛、按区间聚合）
与时序数据不同，混在一起日后不好做归档与清理。
"""

import logging
import os
import sqlite3
import threading
import time

log = logging.getLogger(__name__)

SCHEMA = """
CREATE TABLE IF NOT EXISTS alerts (
    id              INTEGER PRIMARY KEY AUTOINCREMENT,
    rule_id         TEXT    NOT NULL,
    rule_name       TEXT    NOT NULL,
    twin_id         TEXT    NOT NULL,
    level           TEXT    NOT NULL,
    message         TEXT    NOT NULL,
    value           REAL,
    started_at      INTEGER NOT NULL,
    resolved_at     INTEGER,
    acknowledged_at INTEGER,
    acknowledged_by TEXT
);
-- 未恢复的告警要频繁查（前端轮询/推送），单独建索引
CREATE INDEX IF NOT EXISTS idx_active ON alerts (resolved_at);
CREATE INDEX IF NOT EXISTS idx_twin_time ON alerts (twin_id, started_at DESC);
"""


class AlertStore:
    def __init__(self, path="data/alerts.db"):
        self.path = path
        os.makedirs(os.path.dirname(os.path.abspath(path)) or ".", exist_ok=True)
        self._db = sqlite3.connect(path, check_same_thread=False)
        self._db.execute("PRAGMA journal_mode=WAL")
        self._db.executescript(SCHEMA)
        self._db.commit()
        self._lock = threading.Lock()

    # ---- 写入 ----
    def raise_alert(self, rule_id, rule_name, twin_id, level, message, value=None):
        now = int(time.time() * 1000)
        with self._lock:
            cur = self._db.execute(
                "INSERT INTO alerts (rule_id, rule_name, twin_id, level, message, value, started_at)"
                " VALUES (?, ?, ?, ?, ?, ?, ?)",
                (rule_id, rule_name, twin_id, level, message, value, now))
            self._db.commit()
        return {"id": cur.lastrowid, "ruleId": rule_id, "ruleName": rule_name,
                "twinId": twin_id, "level": level, "message": message,
                "value": value, "startedAt": now, "resolvedAt": None,
                "acknowledgedAt": None, "acknowledgedBy": None}

    def resolve(self, alert_id):
        now = int(time.time() * 1000)
        with self._lock:
            self._db.execute(
                "UPDATE alerts SET resolved_at = ? WHERE id = ? AND resolved_at IS NULL",
                (now, alert_id))
            self._db.commit()
        return now

    def acknowledge(self, alert_id, user):
        now = int(time.time() * 1000)
        with self._lock:
            cur = self._db.execute(
                "UPDATE alerts SET acknowledged_at = ?, acknowledged_by = ?"
                " WHERE id = ? AND acknowledged_at IS NULL",
                (now, user, alert_id))
            self._db.commit()
            return cur.rowcount > 0

    # ---- 查询 ----
    def active(self):
        """未恢复的告警（按级别与时间排序：严重的在前）。"""
        with self._lock:
            rows = self._db.execute(
                "SELECT id, rule_id, rule_name, twin_id, level, message, value, started_at,"
                "       resolved_at, acknowledged_at, acknowledged_by"
                " FROM alerts WHERE resolved_at IS NULL"
                " ORDER BY CASE level WHEN 'critical' THEN 0 WHEN 'warning' THEN 1 ELSE 2 END,"
                "          started_at DESC").fetchall()
        return [self._row(r) for r in rows]

    def find_active(self, rule_id, twin_id):
        with self._lock:
            row = self._db.execute(
                "SELECT id FROM alerts WHERE rule_id = ? AND twin_id = ? AND resolved_at IS NULL",
                (rule_id, twin_id)).fetchone()
        return row[0] if row else None

    def history(self, limit=100, twin_id=None, level=None, since=None):
        sql = ("SELECT id, rule_id, rule_name, twin_id, level, message, value, started_at,"
               "       resolved_at, acknowledged_at, acknowledged_by FROM alerts WHERE 1=1")
        args = []
        if twin_id:
            sql += " AND twin_id = ?"
            args.append(twin_id)
        if level:
            sql += " AND level = ?"
            args.append(level)
        if since:
            sql += " AND started_at >= ?"
            args.append(since)
        sql += " ORDER BY started_at DESC LIMIT ?"
        args.append(int(limit))
        with self._lock:
            rows = self._db.execute(sql, args).fetchall()
        return [self._row(r) for r in rows]

    def stats(self, since=None):
        """报表用聚合：按级别、按设备、按规则统计，并给出平均恢复时长。"""
        where = "WHERE started_at >= ?" if since else ""
        args = [since] if since else []

        with self._lock:
            by_level = dict(self._db.execute(
                f"SELECT level, COUNT(*) FROM alerts {where} GROUP BY level", args).fetchall())
            by_twin = [{"twinId": r[0], "count": r[1]}
                       for r in self._db.execute(
                           f"SELECT twin_id, COUNT(*) FROM alerts {where}"
                           " GROUP BY twin_id ORDER BY COUNT(*) DESC", args).fetchall()]
            by_rule = [{"ruleId": r[0], "ruleName": r[1], "count": r[2]}
                       for r in self._db.execute(
                           f"SELECT rule_id, rule_name, COUNT(*) FROM alerts {where}"
                           " GROUP BY rule_id, rule_name ORDER BY COUNT(*) DESC", args).fetchall()]

            total, active_count, acked = self._db.execute(
                f"SELECT COUNT(*), SUM(resolved_at IS NULL), SUM(acknowledged_at IS NOT NULL)"
                f" FROM alerts {where}", args).fetchone()

            # 平均恢复时长：只统计已恢复的
            span_where = f"{where} {'AND' if where else 'WHERE'} resolved_at IS NOT NULL"
            avg_span = self._db.execute(
                f"SELECT AVG(resolved_at - started_at) FROM alerts {span_where}",
                args).fetchone()[0]

        return {
            "total": total or 0,
            "active": active_count or 0,
            "acknowledged": acked or 0,
            "byLevel": by_level,
            "byTwin": by_twin,
            "byRule": by_rule,
            "avgDurationSec": round(avg_span / 1000, 1) if avg_span else None,
        }

    def prune(self, before_ts):
        with self._lock:
            cur = self._db.execute(
                "DELETE FROM alerts WHERE started_at < ? AND resolved_at IS NOT NULL", (before_ts,))
            self._db.commit()
            return cur.rowcount

    @staticmethod
    def _row(r):
        return {
            "id": r[0], "ruleId": r[1], "ruleName": r[2], "twinId": r[3],
            "level": r[4], "message": r[5], "value": r[6],
            "startedAt": r[7], "resolvedAt": r[8],
            "acknowledgedAt": r[9], "acknowledgedBy": r[10],
            "active": r[8] is None,
            "acknowledged": r[9] is not None,
        }

    def close(self):
        try:
            self._db.close()
        except Exception:
            pass

"""进程内数据中枢（Hub）：整个后端唯一的孪生状态源。

所有适配器把 normalize 后的消息经 `ingest` 入站，Hub 负责：
  1. 更新每个孪生对象的最新状态（TwinStore）；
  2. 把消息广播给所有已连接的 WebSocket 客户端（含快照/增量）。

设计要点：
  - 入站 → 落库 → 广播 走同一条路径，天然满足「历史回放」与「断线快照」；
  - 适配器运行在独立线程，Hub 用线程锁保护状态，用 asyncio 队列向 WS 推送。
"""

import asyncio
import threading
import time


class Hub:
    def __init__(self, store=None):
        self._latest = {}        # twinId -> 最新一条孪生消息（dict）
        self._subscribers = {}   # client_id -> asyncio.Queue
        self._loop = None        # WS 服务器所在的事件循环
        self._lock = threading.Lock()
        self._next_client = 0
        self._users = {}         # client_id -> {username, role, since}（在线协同，R10）
        self.store = store       # 可选的时序存储（R9）；为 None 时仅实时
        self.on_ingest = None    # 入站钩子（告警引擎挂在这里）

    # ---- 生命周期（由 ws_server 调用） ----
    def set_loop(self, loop):
        self._loop = loop

    # ---- 入站（适配器调用） ----
    def ingest(self, msg):
        """接收一条已 normalize 的统一孪生消息。"""
        twin_id = msg.get("twinId")
        if not twin_id:
            return
        with self._lock:
            self._latest[twin_id] = msg
        # 落库只进缓冲，不阻塞实时广播
        if self.store:
            self.store.append(msg)
        # 告警评估：钩子内部自己兜异常，不能因为它挂了就断掉实时链路
        if self.on_ingest:
            try:
                self.on_ingest(msg)
            except Exception:  # noqa: BLE001
                pass
        self._broadcast(msg)

    # ---- 出站（WS 客户端） ----
    def snapshot(self):
        """返回所有孪生对象的最新状态列表。"""
        with self._lock:
            return list(self._latest.values())

    def register(self, username=None, role=None):
        """注册一个 WS 客户端，返回 (client_id, asyncio.Queue)。"""
        q = asyncio.Queue(maxsize=1000)
        with self._lock:
            self._next_client += 1
            cid = self._next_client
            self._subscribers[cid] = q
            if username:
                self._users[cid] = {"username": username, "role": role,
                                    "since": int(time.time() * 1000)}
        self.broadcast_presence()
        return cid, q

    def unregister(self, cid):
        with self._lock:
            self._subscribers.pop(cid, None)
            self._users.pop(cid, None)
        self.broadcast_presence()

    def presence(self):
        """在线用户列表。

        同一用户开多个标签页只算一次——否则列表里会出现一串重复的自己。
        """
        with self._lock:
            by_name = {}
            for u in self._users.values():
                by_name[u["username"]] = u
            return sorted(by_name.values(), key=lambda u: u["since"])

    def broadcast_presence(self):
        self._broadcast({
            "type": "presence",
            "ts": int(time.time() * 1000),
            "data": self.presence(),
        })

    def broadcast_event(self, kind, payload):
        """广播一条协同事件（规则变更、操作日志等）。

        与告警一样用**独立信封**：前端据此路由，不塞进 TwinStore。
        """
        self._broadcast({"type": kind, "ts": int(time.time() * 1000), **payload})

    def broadcast_alert(self, action, alert):
        """把告警事件推给所有 WS 客户端。

        用**独立的信封**（没有 twinId）：前端据此路由到告警面板，
        而不是当成孪生消息塞进 TwinStore。
        """
        self._broadcast({
            "type": "alert",
            "action": action,          # fired | resolved
            "ts": int(time.time() * 1000),
            "alert": alert,
        })

    # ---- 内部 ----
    def _broadcast(self, msg):
        if self._loop is None:
            return
        for q in list(self._subscribers.values()):
            # 跨线程安全地把消息投递到 WS 事件循环
            self._loop.call_soon_threadsafe(q.put_nowait, msg)

    try {
      const active = await this.fetchJson('/api/v1/alerts?active=1');
      this.data.active = active.alerts || [];
      this._setBadge(this.data.active.length);

      if (this.tab === 'history') {
        const hist = await this.fetchJson('/api/v1/alerts?limit=100');
        this.data.history = hist.history || [];
      } else if (this.tab === 'report') {
        this.data.stats = await this.fetchJson('/api/v1/alerts/stats');
      }
      this.render();
      return true;
    } catch (err) {
      console.warn('[告警] 拉取失败：', err.message);
      this.body.innerHTML = `<div class="empty">告警数据加载失败：${esc(err.message)}</div>`;
      return false;
    }
  }

  /** 服务端推来的告警事件：刷新列表并更新角标。 */
  onAlert() {
    // 面板开着就立刻刷新；关着只更新角标，避免无谓请求
    if (this.isOpen) this.refresh();
    else this.fetchJson('/api/v1/alerts?active=1')
      .then((r) => this._setBadge((r.alerts || []).length))
      .catch(() => {});
  }

  _setBadge(n) {
    this.activeCount = n;
    this.badge.textContent = String(n);
    this.badge.hidden = n === 0;
  }

  render() {
    if (this.tab === 'active') this.body.innerHTML = this._activeHtml();
    else if (this.tab === 'history') this.body.innerHTML = this._historyHtml();
    else this.body.innerHTML = this._reportHtml();
  }

  _activeHtml() {
    if (!this.data.active.length) {
      return '<div class="empty">当前没有未恢复的告警</div>';
    }
    return this.data.active.map((a) => this._itemHtml(a, false)).join('');
  }

  _historyHtml() {
    if (!this.data.history.length) return '<div class="empty">暂无告警历史</div>';
    return this.data.history.map((a) => this._itemHtml(a, true)).join('');
  }

  _itemHtml(a, withHistory) {
    const resolved = !!a.resolvedAt;
    const dur = a.resolvedAt
      ? fmtDuration(a.resolvedAt - a.startedAt)
      : fmtDuration(Date.now() - a.startedAt);
    const acked = a.acknowledged
      ? `<span class="state">已确认 ${esc(a.acknowledgedBy || '')}</span>`
      : '';
    const ackBtn = (!resolved && !a.acknowledged)
      ? `<button class="ack" data-ack="${a.id}">确认</button>` : '';

    return `
      <div class="alert-item level-${esc(a.level)}${resolved ? ' resolved' : ''}">
        <div class="row1">
          <span class="lvl">${esc(LEVEL_TEXT[a.level] || a.level)}</span>
          <span class="who">${esc(a.twinId)}</span>
          <span class="when">${fmtTime(a.startedAt)}</span>
        </div>
        <div class="msg">${esc(a.message)}</div>
        <div class="foot">
          <span class="state">${resolved ? `已恢复 · 持续 ${dur}` : `进行中 · ${dur}`}</span>
          ${withHistory ? acked : ''}
          ${withHistory ? '' : ackBtn}
        </div>
      </div>`;
  }

  _reportHtml() {
    const s = this.data.stats;
    if (!s) return '<div class="empty">报表加载中…</div>';

    const maxTwin = Math.max(1, ...s.byTwin.map((x) => x.count));
    const maxRule = Math.max(1, ...s.byRule.map((x) => x.count));

    const levelRows = ['critical', 'warning', 'info']
      .map((lv) => `<div class="report-row"><span>${LEVEL_TEXT[lv]}</span>`
        + `<span class="v">${s.byLevel[lv] || 0}</span></div>`)
      .join('');

    const twinRows = s.byTwin.map((x) => `
      <div class="report-row"><span>${esc(x.twinId)}</span><span class="v">${x.count}</span></div>
      <div class="bar"><i style="width:${(x.count / maxTwin) * 100}%"></i></div>`).join('')
      || '<div class="empty">暂无数据</div>';

    const ruleRows = s.byRule.map((x) => `
      <div class="report-row"><span>${esc(x.ruleName)}</span><span class="v">${x.count}</span></div>
      <div class="bar"><i style="width:${(x.count / maxRule) * 100}%"></i></div>`).join('')
      || '<div class="empty">暂无数据</div>';

    return `
      <div class="report-block">
        <h4>总览</h4>
        <div class="report-row"><span>告警总数</span><span class="v">${s.total}</span></div>
        <div class="report-row"><span>当前未恢复</span><span class="v">${s.active}</span></div>
        <div class="report-row"><span>已确认</span><span class="v">${s.acknowledged}</span></div>
        <div class="report-row"><span>平均恢复时长</span><span class="v">${s.avgDurationSec ?? '—'} ${s.avgDurationSec != null ? '秒' : ''}</span></div>
      </div>
      <div class="report-block"><h4>按级别</h4>${levelRows}</div>
      <div class="report-block"><h4>按设备</h4>${twinRows}</div>
      <div class="report-block"><h4>按规则</h4>${ruleRows}</div>`;
  }
}

function esc(s) {
  return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
}

const probeItem = st.items.find((i) => i.text.includes(PROBE));
if (probeItem) {
  check(probeItem.level === 'critical', `级别为 ${probeItem.level}（规则里配的是 critical）`);
  check(!probeItem.resolved, '状态为「进行中」而非已恢复');
  check(probeItem.hasAck, '提供了「确认」按钮');
}
check(!st.badgeHidden && Number(st.badgeText) >= 1,
  `角标显示未恢复告警数：${st.badgeText}`);

// ---- ④ 未读数随 WS 推送更新（面板关闭时也要更新角标）----
console.log('\n=== ④ 关闭面板后角标仍随推送更新 ===');
await page.click('#alertClose');
await sleep(500);
const closedState = await panelState();
check(!closedState.open, '抽屉已关闭');
check(!closedState.badgeHidden, `关闭状态下角标仍在：${closedState.badgeText}`);

// ---- ⑤ 确认告警 ----
console.log('\n=== ⑤ 确认告警 ===');
await page.click('#btnAlerts');
await sleep(1000);
const ackResult = await page.evaluate(async (probe) => {
  const item = [...document.querySelectorAll('.alert-item')]
    .find((el) => el.textContent.includes(probe));
  const btn = item?.querySelector('button[data-ack]');
  if (!btn) return { clicked: false };
  const id = btn.dataset.ack;
  btn.click();
  await new Promise((r) => setTimeout(r, 1500));
  const api = await fetch(`/api/v1/alerts?limit=50`, {
    headers: { Authorization: `Bearer ${sessionStorage.getItem('twin-session-token')}` },
  }).then((r) => r.json());
  const rec = (api.history || []).find((a) => a.id === Number(id));
  return { clicked: true, id, acknowledged: !!rec?.acknowledged, by: rec?.acknowledgedBy };
}, PROBE);

check(ackResult.clicked, '找到并点击了确认按钮');
check(ackResult.acknowledged, `服务端记录了确认（id=${ackResult.id}，by=${ackResult.by}）`);

// 确认后按钮应消失（已确认的不再显示确认按钮）
await sleep(800);
await page.click('.tab[data-tab="history"]');
await sleep(900);
const afterAck = await panelState();
const ackedItem = afterAck.items.find((i) => i.text.includes(PROBE));
check(ackedItem && !ackedItem.hasAck, '确认后不再显示确认按钮');
check(ackedItem && /已确认/.test(ackedItem.text), '历史里标注了确认人');

mkdirSync(join(root, 'tools', 'render-checks'), { recursive: true });
await page.click('.tab[data-tab="active"]');
await sleep(400);
await page.screenshot({ path: join(root, 'tools', 'render-checks', 'alerts.png') });
console.log('  截图 → tools/render-checks/alerts.png');

await browser.close();

console.log('\n=== 运行时问题 ===');
if (problems.length === 0) console.log('  ✓ 无控制台错误');
else { for (const p of problems) console.log('  ✗ ' + p); errors += problems.length; }

console.log('\n' + (errors === 0 ? '✓ 告警面板校验通过' : `✗ ${errors} 项失败`));
process.exit(errors ? 1 : 0);
